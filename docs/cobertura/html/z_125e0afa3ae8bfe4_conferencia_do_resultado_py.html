83def conferir_resultado(dados: object) -> ResultadoDoPentest: 

84 campos = exigir_campos(dados, CAMPOS_DO_RESULTADO, "") 

85 exigir_versao(campos)

107def converter_validacoes(campos: dict[str, Any]) -> list[ValidacaoDoPentest]: 

108 itens = converter_lista(campos, "validacoes", "") 

109 if len(itens) > TAMANHO_MAXIMO_DE_VALIDACOES: 

110 raise CampoForaDoFormato(f"`validacoes` passa de {TAMANHO_MAXIMO_DE_VALIDACOES} itens") 

111 validacoes = [] 

112 for indice, item in enumerate(itens): 

113 validacoes.append(converter_validacao(item, f"validacoes[{indice}]")) 

114 exigir_identificadores_sem_repeticao(validacoes) 

115 return validacoes

238def exigir_campos( 

239 objeto: object, esperados: frozenset[str], caminho_do_campo: str 

240) -> dict[str, Any]: 

241 if not isinstance(objeto, dict): 

242 raise CampoForaDoFormato(f"{descrever_objeto(caminho_do_campo)} precisa ser um objeto JSON") 

243 a_mais = sorted(set(objeto) - esperados) 

244 faltando = sorted(esperados - set(objeto)) 

245 if a_mais: 

246 raise CampoForaDoFormato( 

247 f"campo a mais {local_do_campo(caminho_do_campo)}: {', '.join(a_mais)}" 

248 ) 

249 if faltando: 

250 raise CampoForaDoFormato( 

251 f"campo faltando {local_do_campo(caminho_do_campo)}: {', '.join(faltando)}" 

252 ) 

253 return objeto

279def converter_texto(campos: dict[str, Any], campo: str, caminho_do_campo: str, tamanho: int) -> str: 

280 valor = campos[campo] 

281 if not isinstance(valor, str) or not valor.strip(): 

282 raise CampoForaDoFormato(f"`{caminho_do_campo}.{campo}` precisa ser um texto não vazio") 

283 texto = valor.strip() 

284 if len(texto) > tamanho: 

285 raise CampoForaDoFormato(f"`{caminho_do_campo}.{campo}` passa de {tamanho} caracteres") 

286 if parece_trazer_segredo(texto): 

287 raise CampoForaDoFormato(f"`{caminho_do_campo}.{campo}` parece trazer um segredo") 

288 return texto

331def converter_url(campos: dict[str, Any], caminho_do_campo: str) -> str: 

332 url = campos["url"] 

333 if not isinstance(url, str) or not url.strip(): 

334 raise CampoForaDoFormato(f"`{caminho_do_campo}.url` precisa ser uma URL") 

335 try: 

336 partes = urlsplit(url) 

337 except ValueError as erro: 

338 raise CampoForaDoFormato(f"`{caminho_do_campo}.url` não é uma URL válida") from erro 

339 if not url_http_com_host(partes): 

340 raise CampoForaDoFormato(f"`{caminho_do_campo}.url` precisa ser uma URL http ou https") 

341 if traz_query_fragmento_ou_credenciais(partes): 

342 raise CampoForaDoFormato( 

343 f"`{caminho_do_campo}.url` não pode trazer query, fragmento ou credenciais" 

344 ) 

345 return url

348def url_http_com_host(partes: SplitResult) -> bool: 

349 return partes.scheme.lower() in ESQUEMAS_DA_WEB and bool(partes.netloc)

352def traz_query_fragmento_ou_credenciais(partes: SplitResult) -> bool: 

353 partes_proibidas = (partes.query, partes.fragment, partes.username, partes.password) 

354 return any(partes_proibidas)
