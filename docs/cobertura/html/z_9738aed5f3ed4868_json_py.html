80def listar_limitacoes(analise: Analise) -> list[dict[str, str]]: 

81 limitacoes = [] 

82 if analise.provedor_de_ia is None: 

83 limitacoes.append(limitacao("bundles_nao_procurados", TEXTO_BUNDLES_NAO_PROCURADOS)) 

84 limitacoes.append(limitacao("achados_sem_explicacao", TEXTO_ACHADOS_SEM_EXPLICACAO)) 

85 if not analise.tem_javascript: 

86 limitacoes.append(limitacao("sem_javascript", TEXTO_SEM_JAVASCRIPT)) 

87 limitacoes.extend(limitacoes_da_ia_que_falhou(analise)) 

88 for fonte in fontes_que_nao_responderam(analise.fontes_sem_resposta): 

89 texto = texto_da_fonte_sem_resposta(fonte, analise.fontes_sem_resposta) 

90 limitacoes.append(limitacao(f"fonte_sem_resposta_{fonte.lower()}", texto)) 

91 return limitacoes

94def limitacoes_da_ia_que_falhou(analise: Analise) -> list[dict[str, str]]: 

95 limitacoes = [] 

96 for etapa, (identificador, consequencia) in CONSEQUENCIA_DA_IA_QUE_FALHOU.items(): 

97 chamadas = [chamada for chamada in analise.chamadas_de_ia if chamada.etapa == etapa] 

98 falhas = [chamada for chamada in chamadas if not chamada.sucesso] 

99 if falhas: 

100 texto = ( 

101 f"A IA ({analise.provedor_de_ia}) falhou em {len(falhas)} de {len(chamadas)} " 

102 f"chamadas para {consequencia} O motivo está no painel da IA." 

103 ) 

104 limitacoes.append(limitacao(identificador, texto)) 

105 return limitacoes

164def biblioteca_no_relatorio(biblioteca: Biblioteca, analise: Analise) -> dict[str, Any]: 

165 chave = chave_da_biblioteca(biblioteca) 

166 achados = achados_da_biblioteca(chave, analise.achados) 

167 explicacao = analise.explicacoes.get(chave) 

168 nota = nota_da_biblioteca(achados) 

169 return { 

170 "id": chave, 

171 **asdict(replace(biblioteca, evidencias=evidencias_sem_valores(biblioteca.evidencias))), 

172 "nota": nota, 

173 "prioridade": classificar_prioridade(nota) if achados else None, 

174 "achados": [achado.aviso.ghsa for achado in achados], 

175 "motivo_nao_consultada": analise.nao_consultadas.get(chave), 

176 "explicacao": asdict(explicacao) if explicacao is not None else None,

215def comparar_com_e_sem_ia(analise: Analise) -> dict[str, dict[str, int]] | None: 

216 if analise.provedor_de_ia is None: 

217 return None 

218 achados_por_assinatura = [ 

219 achado for achado in analise.achados if achado.biblioteca.metodo == "assinatura" 

220 ] 

221 bibliotecas_por_assinatura = [ 

222 biblioteca for biblioteca in analise.bibliotecas if biblioteca.metodo == "assinatura"

224 return { 

225 "bibliotecas": contagem(len(bibliotecas_por_assinatura), len(analise.bibliotecas)), 

226 "achados": contagem(len(achados_por_assinatura), len(analise.achados)),

259def agrupar_bibliotecas_por_requisicao(bibliotecas: list[Biblioteca]) -> dict[int, list[str]]: 

260 agrupadas: dict[int, list[str]] = {} 

261 for biblioteca in bibliotecas: 

262 for indice in {evidencia.indice_requisicao for evidencia in biblioteca.evidencias}: 

263 agrupadas.setdefault(indice, []).append(chave_da_biblioteca(biblioteca)) 

264 return agrupadas
